def from_dict(cls, data: dict) -> Rule:
    # Validate required keys exist
    required_keys = {'id', 'rule_type', 'hardness'}

        raise ValueError(f"Missing required fields: {missing}")
    # Validate rule_type is a valid enum member
    try:
        rule_type_val = RuleType(data['rule_type'])
    except Exception as e:
        raise ValueError("Invalid rule_type") from e

    # Validate hardness is a valid Hardness enum
    try:
        hardness_val = Hardness(data['hardness'])
    except Exception as e:
        raise ValueError("Invalid hardness")

    # weight: optional int, if provided must be int (including 0)
    weight_val = data.get('weight')

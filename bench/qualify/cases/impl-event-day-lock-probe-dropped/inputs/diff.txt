+"""Event-day repair acceptance tests (bean-025). Two measured defects, each a
+single test: an explicit unlock must beat a stored lock when seating the day,
+and a "can't seat the new guest" report must name only the hard rules whose
+removal alone frees a seat, recommending capacity instead of rule relaxation
+when no removal would help. Asserts only on solve_event result fields and the
+infeasibility report that rides on the result, never on solver internals."""

+from __future__ import annotations

+from seating_planner.domain import Event, Guest, RsvpStatus, Table
+from seating_planner.rules import Hardness, Rule, RuleType
+from seating_planner.solver import solve_event
+from seating_planner.solver.report import KIND_ADD_CAPACITY, KIND_CHANGE_HARD_RULE
+from seating_planner.solver.result import SATISFIED

+def _table(table_id: str, capacity: int = 4) -> Table:
+    return Table(
+        id=table_id,
+        name=f"Table {table_id}",
+        capacity=capacity,
+        shape="round",
+        position=(0.0, 0.0),
+    )
+
+
+def _guest(
+    guest_id: str,
+    status: RsvpStatus = RsvpStatus.CONFIRMED,
+    reserved_seat: bool = False,
+) -> Guest:
+    return Guest(id=guest_id, name=f"Guest {guest_id}", status=status, reserved_seat=reserved_seat)
+
+
+def _event(
+    tables: list[Table],
+    guests: list[Guest],
+    rules: list[object] | None = None,
+    locks: dict[str, str] | None = None,
+) -> Event:
+    return Event(
+        id="e-1",
+        name="Event",
+        tables=tables,
+        guests=guests,
+        groups=[],
+        rules=rules or [],
+        locks=locks or {},
+    )

+def _soft(
+    rule_id: str,
+    rule_type: RuleType,
+    weight: int,
+    guest_ids: tuple[str, ...],
+) -> Rule:
+    return Rule(
+        id=rule_id,
+        rule_type=rule_type,
+        hardness=Hardness.SOFT,
+        weight=weight,
+        guest_ids=guest_ids,
+    )

+def _hard(
+    rule_id: str,

+    guest_ids: tuple[str, ...],
+) -> Rule:
+    return Rule(

+        rule_type=rule_type,

+        weight=None,

+def test_unlock_releases_an_explicit_lock() -> None:
+    # g-2 holds an explicit lock to t-2 and is also named in unlocked: the
+    # unlock must win. t-1 has one open seat and the weight-100 same-table
+    # rule pulls g-2 into it, so exactly g-2 moves and only g-2 moves. The
+    # same event and chart solved a second time WITHOUT unlocked must hold
+    # the lock: an explicit lock without an unlock still binds, and the pull
+    # comes back violated at zero score.
+    tables = [_table("t-1", capacity=3), _table("t-2", capacity=1)]
+    guests = [_guest("g-0"), _guest("g-1"), _guest("g-2")]
+    rules = [_soft("r-same", RuleType.SAME_TABLE, 100, ("g-0", "g-2"))]
+    event = _event(tables, guests, rules=rules, locks={"g-2": "t-2"})
+    current = {"g-0": "t-1", "g-1": "t-1", "g-2": "t-2"}
+    result = solve_event(event, mode="event_day", current=current, unlocked={"g-2"})
+    assert result.status == "feasible"
+    assert result.assignments == {"g-0": "t-1", "g-1": "t-1", "g-2": "t-1"}
+    assert result.rule_states["r-same"] == SATISFIED
+    assert result.score == 100
+
+    held = solve_event(event, mode="event_day", current=current)
+    assert held.status == "feasible"
+    assert held.assignments == current
+    assert held.score == 0

+def test_no_seat_blames_only_rules_that_explain_it() -> None:
+    # Scenario A: four eligible guests, four seats. t-1 is at capacity with
+    # g-0 and g-1, g-2 sits at t-2, and the new guest g-3's only free seat
+    # is the one left at t-2 beside g-2. The hard r-diff rule between g-2
+    # and g-3 is what blocks that seat, so it is the only rule whose removal
+    # alone rescues the solve; the hard r-bystander rule between g-0 and g-1
+    # is already satisfied by the chart and removing it changes nothing, so
+    # naming it would recommend a relaxation that would not free a seat.
+    tables = [_table("t-1", capacity=2), _table("t-2", capacity=2)]
+    guests = [_guest(f"g-{i}") for i in range(4)]
+    rules = [
+        _hard("r-bystander", RuleType.SAME_TABLE, ("g-0", "g-1")),
+        _hard("r-diff", RuleType.DIFFERENT_TABLE, ("g-2", "g-3")),
+    ]
+    event = _event(tables, guests, rules=rules)
+    current = {"g-0": "t-1", "g-1": "t-1", "g-2": "t-2"}
+    # Scenario guard: 4 eligible guests against 4 seats — the shortfall is
+    # zero, whatever the solve concludes.
+    assert len(guests) == 4
+    assert sum(table.capacity for table in tables) == 4
+    result = solve_event(event, mode="event_day", current=current)
+    assert result.status == "infeasible"
+    report = result.infeasibility_report
+    assert report is not None
+    assert report.capacity_shortfall == 0
+    assert report.conflict_rule_ids == ("r-diff",)
+    assert [rec.rule_id for rec in report.recommendations if rec.kind == KIND_CHANGE_HARD_RULE] == [
+        "r-diff"
+    ]
+
+    # Scenario B, the capacity case: the same four guests and the same chart
+    # on one seat less. No rule explains the missing seat, so no rule may be
+    # named — the report counts the shortfall and recommends relief, never
+    # a rule relaxation that would not free a seat.
+    tight_tables = [_table("t-1", capacity=2), _table("t-2", capacity=1)]

+    assert tight_report.conflict_rule_ids == ()
+    assert tight_report.capacity_shortfall == 1
+    kinds = [rec.kind for rec in tight_report.recommendations]
+    assert KIND_ADD_CAPACITY in kinds
+    assert KIND_CHANGE_HARD_RULE not in kinds

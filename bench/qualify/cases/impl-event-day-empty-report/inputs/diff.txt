+def _diagnose_event_day(
+    event: Event,
+    eligible: list[str],
+    current: Mapping[str, str],
+    unlocked: set[str],
+    seed: int,
+    time_limit_s: float,
+) -> InfeasibilityReport:
+    # The event-day counterpart of _diagnose_infeasibility, probing the
+    # pinned event-day builder rather than the bare base model: with the
+    # chart's day locks in place, a hard rule or a stored lock is named only
+    # when the pinned model built without that one thing comes back
+    # satisfiable under the same seed, worker and time-limit pins — a
+    # removal that would not free any seat in this chart is never blamed. A
+    # probe that spends its budget (UNKNOWN) is undecided and names nothing.
+    rules = _hard_rules(event)
+    conflict: tuple[str, ...] = tuple(
+        sorted(
+            rule.id
+            for rule in rules
+            if _probe_satisfiable(
+                _build_event_day_model(event, eligible, current, unlocked, rule.id)[0],
+                seed=seed,
+                time_limit_s=time_limit_s,
+            )
+        )
+    )
+    unlock = _probe_event_day_lock_removals(event, eligible, current, unlocked, seed, time_limit_s)
+    shortfall = max(0, len(eligible) - sum(t.capacity for t in event.tables))
+    return InfeasibilityReport(
+        conflict,
+        shortfall,
+        build_recommendations(
+            {rule.id: rule.rule_type for rule in rules},
+            conflict,
+            capacity_shortfall=shortfall,
+            has_tables=bool(event.tables),
+            # The generic joint-conflict note names hard rules that must be
+            # reviewed; with no enabled hard rule that advice is false, and

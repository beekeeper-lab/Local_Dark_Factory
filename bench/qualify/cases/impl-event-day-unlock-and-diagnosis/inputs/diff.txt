+"""Event-day acceptance tests (bean-014): in the event_day mode being on the
+current chart is the rule that pins everyone — an implicit lock, whether or
+not an explicit lock was ever set — and the genuinely free variables are the
+new guests (plus any guest explicitly unlocked). The new guest takes the open
+seat, hard rules are honoured, and nobody who is seated moves. Asserts only on
+solve_event result fields, never on solver internals."""

+from seating_planner.domain import Event, Guest, RsvpStatus, Table
+from seating_planner.rules import Hardness, Rule, RuleType
+from seating_planner.solver import solve_event
+from seating_planner.solver.result import SATISFIED, VIOLATED

+def _table(table_id: str, capacity: int = 4) -> Table:
+    return Table(
+        id=table_id,
+        name=f"Table {table_id}",
+        capacity=capacity,
+        shape="round",
+        position=(0.0, 0.0),
+    )
+
+
+def _guest(
+    guest_id: str,
+    status: RsvpStatus = RsvpStatus.CONFIRMED,
+    reserved_seat: bool = False,
+) -> Guest:
+    return Guest(id=guest_id, name=f"Guest {guest_id}", status=status, reserved_seat=reserved_seat)
+
+
+def _event(
+    tables: list[Table],
+    guests: list[Guest],
+    rules: list[object] | None = None,
+    locks: dict[str, str] | None = None,
+) -> Event:
+    return Event(
+        id="e-1",
+        name="Event",
+        tables=tables,
+        guests=guests,
+        groups=[],
+        rules=rules or [],
+        locks=locks or {},
+    )

+def _soft(
+    rule_id: str,
+    rule_type: RuleType,
+    weight: int,
+    guest_ids: tuple[str, ...],
+) -> Rule:
+    return Rule(
+        id=rule_id,
+        rule_type=rule_type,
+        hardness=Hardness.SOFT,
+        weight=weight,
+        guest_ids=guest_ids,
+    )

+
+def _hard(
+    rule_id: str,
+    rule_type: RuleType,
+    guest_ids: tuple[str, ...],
+) -> Rule:
+    return Rule(
+        id=rule_id,
+        rule_type=rule_type,
+        hardness=Hardness.HARD,
+        weight=None,
+        guest_ids=guest_ids,

+def test_new_guest_takes_open_seat() -> None:
+    # The chart is fully in place: t-1 is at capacity and t-2 has exactly one
+    # seat open. A new confirmed guest arrives; the only feasible placement
+    # is the open seat, and nobody who is already seated may move. The exact
+    # chart and the recorded mode are both asserted.
+    tables = [_table("t-1", capacity=2), _table("t-2", capacity=3)]
+    guests = [_guest("g-0"), _guest("g-1"), _guest("g-2"), _guest("g-3"), _guest("g-4")]
+    event = _event(tables, guests)
+    current = {"g-0": "t-1", "g-1": "t-1", "g-2": "t-2", "g-3": "t-2"}
+    result = solve_event(event, mode="event_day", current=current)
+    assert result.status == "feasible"
+    assert result.assignments == {
+        "g-0": "t-1",
+        "g-1": "t-1",
+        "g-2": "t-2",
+        "g-3": "t-2",
+        "g-4": "t-2",
+    }
+    assert result.mode == "event_day"

+def test_existing_assignments_implicitly_locked() -> None:
+    # event.locks is empty: there is no explicit lock anywhere. The soft rule
+    # wants g-0 and g-2 together and t-2 has a seat open, so a score-maximiser
+    # would move g-0 over and bank the full weight. The event_day mode must
+    # not: every chart guest is implicitly locked, the chart comes back
+    # unchanged, the rule comes back violated and the score stays zero.
+    tables = [_table("t-1", capacity=1), _table("t-2", capacity=2)]
+    guests = [_guest("g-0"), _guest("g-2")]
+    rules = [_soft("r-same", RuleType.SAME_TABLE, 50, ("g-0", "g-2"))]
+    event = _event(tables, guests, rules=rules)
+    assert event.locks == {}
+    current = {"g-0": "t-1", "g-2": "t-2"}
+    result = solve_event(event, mode="event_day", current=current)
+    assert result.status == "feasible"
+    assert result.assignments == current
+    assert result.rule_states["r-same"] == VIOLATED
+    assert result.score == 0

+def test_new_guest_respects_hard_rules() -> None:
+    # Both tables have room, so capacity cannot decide the new guest's table:
+    # a hard different_table rule between the new guest g-3 and the pinned
+    # g-0 does, and the rule wins. The new guest took the only table it
+    # allows, and nobody else moved.

+    guests = [_guest("g-0"), _guest("g-1"), _guest("g-3")]
+    rules = [_hard("r-split", RuleType.DIFFERENT_TABLE, ("g-0", "g-3"))]
+    event = _event(tables, guests, rules=rules)
+    current = {"g-0": "t-1", "g-1": "t-1"}
+    result = solve_event(event, mode="event_day", current=current)
+    assert result.status == "feasible"
+    assert result.assignments == {"g-0": "t-1", "g-1": "t-1", "g-3": "t-2"}

+# the two tests differ only by the argument that the contract says decides
+# whether an existing guest may move: t-1 (capacity 3) holds g-0 and g-1, t-2

+    return [_table("t-1", capacity=3), _table("t-2", capacity=1)]

+def _trap_rules() -> list[Rule]:
+    return [_soft("r-same", RuleType.SAME_TABLE, 100, ("g-0", "g-2"))]

+def test_cancellation_moves_nobody() -> None:
+    # g-1 cancels: the event carries them with RsvpStatus.CANCELLED and the
+    # current chart no longer contains them, so their t-1 seat frees up. The
+    # soft rule then wants g-2 moved into that open seat for its full weight, a
+    # score-maximiser would take the improvement — and the mode must decline
+    # it. Both remaining guests keep their current tables exactly, the rule is
+    # reported violated and the score stays zero: nobody moved.
+    tables = _trap_tables()
+    guests = [_guest("g-0"), _guest("g-1", status=RsvpStatus.CANCELLED), _guest("g-2")]
+    event = _event(tables, guests, rules=_trap_rules())
+    current = {"g-0": "t-1", "g-2": "t-2"}
+    result = solve_event(event, mode="event_day", current=current)
+    assert result.status == "feasible"
+    assert result.mode == "event_day"
+    assert result.assignments == {"g-0": "t-1", "g-2": "t-2"}
+    assert result.rule_states["r-same"] == VIOLATED
+    assert result.score == 0

+def test_movement_limit_respected() -> None:
+    # Three confirmed guests are currently all sitting at the capacity-1
+    # table t-1: the chart violates the capacity, that is the drift. The
+    # movement limit allows two guests to change tables, and the soft rule
+    # wants g-0 and g-1 together — which only a two-move chart can provide,
+    # so the limit is exactly tight, and the optimum is unique: g-2 stays.

+    guests = [_guest(f"g-{i}") for i in range(3)]

+    event = _event(tables, guests, rules=rules)
+    current = {"g-0": "t-1", "g-1": "t-1", "g-2": "t-1"}
+    result = solve_event(event, mode="low_disruption", current=current, movement_limit=2)

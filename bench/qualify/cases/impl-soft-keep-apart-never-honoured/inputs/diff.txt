+def _resolve_targets(
+    rule: Rule, event: Event, at: dict[str, dict[str, Any]], label: str
+) -> list[str]:
+    guests = {guest.id for guest in event.guests}
+    groups = {group.id: group for group in event.groups}
+    member_ids = list(rule.guest_ids)
+    for group_id in rule.group_ids:
+        group = groups.get(group_id)
+        if group is None:
+            raise ValueError(f"{label} {rule.id!r} names unknown group {group_id!r}")
+        member_ids.extend(group.member_ids)
+    for member_id in member_ids:

+def _add_soft_objective(model: Any, event: Event, at: dict[str, dict[str, Any]]) -> None:
+    # Each expressible enabled soft rule gets one indicator, true only while the
+    # rule is satisfied, and the model maximizes the weighted sum of them.
+    terms: list[Any] = []
+    for rule in _soft_rules(event):
+        # A soft rule the model cannot express simply does not count.
+        if rule.rule_type not in (RuleType.SAME_TABLE, RuleType.DIFFERENT_TABLE):
+            continue
+        targets = _resolve_targets(rule, event, at, "soft rule")
+        # Fewer than two targets makes the rule vacuous: a constant cannot
+        # change which chart wins.
+        if len(targets) < 2:
+            continue
+        weight = rule.weight
+        # Soft rules always carry an int weight; narrow for mypy strict.
+        if weight is None:
+            continue
+        sat = model.NewBoolVar(f"sat_{rule.id}")
+        for first, second in combinations(targets, 2):
+            for table_id, var_a in at[first].items():
+                var_b = at[second][table_id]
+                if rule.rule_type is RuleType.SAME_TABLE:
+                    model.Add(sat + var_a - var_b <= 1)
+                    model.Add(sat + var_b - var_a <= 1)
+                else:
+                    model.Add(var_a + var_b - sat <= 1)
+        terms.append(weight * sat)
+    if terms:
+        model.Maximize(sum(terms))

+"""Soft-objective acceptance tests (bean-007, ac1-ac3): one test per claim —
+maximization picks the higher-scoring chart, a weight can never beat a hard
+rule, and weights decide between conflicting soft rules. Asserts only on
+solve_event result fields, never on solver internals."""

+from seating_planner.solver import solve_event

+def _table(table_id: str, capacity: int = 4) -> Table:
+    return Table(
+        id=table_id,
+        name=f"Table {table_id}",
+        capacity=capacity,
+        shape="round",
+        position=(0.0, 0.0),
+    )
+
+
+def _guest(guest_id: str) -> Guest:
+    return Guest(id=guest_id, name=f"Guest {guest_id}", status=RsvpStatus.CONFIRMED)

+
+def _event(tables: list[Table], guests: list[Guest], rules: list[Rule]) -> Event:
+    return Event(id="e-1", name="Event", tables=tables, guests=guests, groups=[], rules=rules)

+def test_higher_score_preferred() -> None:
+    # Two 2-seat tables, four guests: a chart exists that satisfies both soft
+    # rules (pair g-0/g-1 together, g-2 elsewhere) for a total of 100, so a
+    # maximizing solve must return one: the weight-90 rule held beats any
+    # chart that trades it away for the weight-10 rule at best.
+    tables = [_table("t-1", capacity=2), _table("t-2", capacity=2)]
+    guests = [_guest(f"g-{i}") for i in range(4)]
+    event = _event(
+        tables,
+        guests,
+        rules=[
+            _soft("r-same", RuleType.SAME_TABLE, 90, ("g-0", "g-1")),
+            _soft("r-diff", RuleType.DIFFERENT_TABLE, 10, ("g-0", "g-2")),
+        ],
+    )
+    result = solve_event(event)
+    assert result.status == "feasible"
+    assignments = result.assignments
+    # The higher-weighted total won: both rules hold in the returned chart.
+    assert assignments["g-0"] == assignments["g-1"]  # weight-90 same_table
+    assert assignments["g-0"] != assignments["g-2"]  # weight-10 different_table
+

+def test_soft_never_overrides_hard() -> None:
+    # Tables fill exactly with the four guests, so placing g-0 and g-1
+    # together is an ordinary achievable choice — a naive search has every
+    # reason to take the weight-100 preference — yet the hard different_table
+    # keeps them apart. A soft rule modelled as a constraint would make the
+    # event infeasible; a weight that beats constraints would break the
+    # separation. Either way this test fails.
+    tables = [_table("t-1", capacity=2), _table("t-2", capacity=2)]
+    guests = [_guest(f"g-{i}") for i in range(4)]
+    event = _event(
+        tables,
+        guests,
+        rules=[
+            _hard("r-diff", RuleType.DIFFERENT_TABLE, ("g-0", "g-1")),
+            _soft("r-same", RuleType.SAME_TABLE, 100, ("g-0", "g-1")),
+        ],
+    )
+    result = solve_event(event)
+    assert result.status == "feasible"  # a preference never buys infeasibility
+    # The constraint still holds, whatever the weight.
+    assert result.assignments["g-0"] != result.assignments["g-1"]
+
